from typing import Any

from beanie import Document, init_beanie
from bson import ObjectId
from fastapi import Depends, FastAPI, File, HTTPException, UploadFile
from fastapi.responses import StreamingResponse
from gridfs import GridFSBucket
from motor.motor_asyncio import AsyncIOMotorClient, AsyncIOMotorDatabase
from pymongo.database import Database  # Explicitly import Database[Any]


# Define a Beanie model (optional)
class MyDocument(Document):
    name: str

    class Settings:
        collection = "my_collection"


# Database initialization function
async def init_db() -> tuple[AsyncIOMotorDatabase, AsyncIOMotorClient]:
    """Initialize MongoDB and Beanie, returning (db, client)."""
    client: AsyncIOMotorClient = AsyncIOMotorClient(
        "mongodb://localhost:27017")
    db = client["my_database"]  # Correctly getting the database

    # Initialize Beanie with models
    await init_beanie(database=db, document_models=[MyDocument])

    return db, client


# FastAPI App with `lifespan()` for DB Lifecycle Management
async def lifespan(app: FastAPI):
    """Manages the application's lifespan, handling DB connection."""
    db, client = await init_db()
    app.state.db = db  # Store the db in app state
    app.state.client = client
    yield  # Wait until shutdown
    client.close()  # Proper cleanup


app = FastAPI(lifespan=lifespan)


# ✅ Corrected DB Dependency Function
def get_db() -> AsyncIOMotorDatabase:
    """Provides the MongoDB database instance from app state."""
    db: AsyncIOMotorDatabase = app.state.db  # Explicitly retrieving as AsyncIOMotorDatabase
    return db


# Upload a file to GridFS
@app.post("/upload/")
async def upload_file(file: UploadFile = File(...), db: Database[Any] = Depends(get_db)):
    if file.filename is None:
        raise HTTPException(status_code=400, detail="No file provided")

    fs = GridFSBucket(db)  # Now correctly recognized as Database[Any]
    file_id = fs.upload_from_stream(file.filename, await file.read())
    return {"filename": file.filename, "file_id": str(file_id)}


# Retrieve a file from GridFS
@app.get("/download/{file_id}")
async def download_file(file_id: str, db: Database[Any] = Depends(get_db)):
    fs = GridFSBucket(db)

    try:
        file_stream = await fs.open_download_stream(ObjectId(file_id))
    except Exception:
        raise HTTPException(status_code=404, detail="File not found")

    async def file_iterator():
        while chunk := await file_stream.read(1024):
            yield chunk

    return StreamingResponse(file_iterator(), media_type="application/octet-stream")


# Delete a file from GridFS
@app.delete("/delete/{file_id}")
async def delete_file(file_id: str, db: Database[Any] = Depends(get_db)):
    fs = GridFSBucket(db)

    try:
        fs.delete(ObjectId(file_id))
    except Exception:
        raise HTTPException(status_code=404, detail="File not found")

    return {"message": "File deleted successfully"}
